// Variables globales para la sesión de Drive let tokenClient; let accessToken = null; let rootFolderId = null; // Inicialización de Google Identity Services function gapiLoaded() { tokenClient = google.accounts.oauth2.initTokenClient({ client_id: 'TU_CLIENT_ID_DE_GOOGLE.apps.googleusercontent.com', // Reemplaza con tu Client ID real scope: 'https://www.googleapis.com/auth/drive.file', callback: async (resp) => { if (resp.error !== undefined) { console.error("Error de autenticación:", resp); return; } accessToken = resp.access_token; document.getElementById('drive-status').innerText = 'Conectado a Google Drive. Verificando estructura...'; document.getElementById('btn-auth').style.display = 'none'; document.getElementById('btn-signout').style.display = 'inline-block'; // Inicializar la carpeta raíz al conectar await initializeDriveStructure(); }, }); } function handleAuthClick() { if (tokenClient) { tokenClient.requestAccessToken({ prompt: 'consent' }); } } function handleSignoutClick() { if (accessToken) { google.accounts.oauth2.revoke(accessToken, () => { accessToken = null; rootFolderId = null; document.getElementById('drive-status').innerText = 'Desconectado de Google Drive.'; document.getElementById('btn-auth').style.display = 'inline-block'; document.getElementById('btn-signout').style.display = 'none'; }); } } // Buscar o crear la carpeta raíz "Armadillo Centro" async function initializeDriveStructure() { try { // 1. Buscar si ya existe la carpeta raíz const query = "name = 'Armadillo Centro' and mimeType = 'application/vnd.google-apps.folder' and trashed = false"; const response = await fetch(`https://www.googleapis.com/drive/v3/files?q=${encodeURIComponent(query)}`, { headers: { Authorization: `Bearer ${accessToken}` } }); const data = await response.json(); if (data.files && data.files.length > 0) { rootFolderId = data.files[0].id; document.getElementById('drive-status').innerText = 'Conectado. Carpeta "Armadillo Centro" localizada.'; } else { // 2. Si no existe, crear la carpeta raíz const folderMetadata = { name: 'Armadillo Centro', mimeType: 'application/vnd.google-apps.folder' }; const createResponse = await fetch('https://www.googleapis.com/drive/v3/files', { method: 'POST', headers: { Authorization: `Bearer ${accessToken}`, 'Content-Type': 'application/json' }, body: JSON.stringify(folderMetadata) }); const folder = await createResponse.json(); rootFolderId = folder.id; // Crear subcarpetas esenciales (proyectos, planes, analisis, adjuntos) await createSubFolders(rootFolderId); document.getElementById('drive-status').innerText = 'Conectado. Carpeta "Armadillo Centro" creada con éxito.'; } } catch (error) { console.error("Error al inicializar la estructura en Drive:", error); document.getElementById('drive-status').innerText = 'Error al sincronizar carpetas en Drive.'; } } async function createSubFolders(parentId) { const subfolders = ['proyectos', 'planes', 'analisis', 'adjuntos']; for (const name of subfolders) { const metadata = { name: name, mimeType: 'application/vnd.google-apps.folder', parents: [parentId] }; await fetch('https://www.googleapis.com/drive/v3/files', { method: 'POST', headers: { Authorization: `Bearer ${accessToken}`, 'Content-Type': 'application/json' }, body: JSON.stringify(metadata) }); } } // Función para guardar un archivo Markdown con Frontmatter en Drive async function saveMarkdownFile(subfolderName, fileName, markdownContent) { if (!rootFolderId || !accessToken) { alert("No estás conectado a Google Drive o no se ha inicializado la carpeta raíz."); return; } try { // Buscar ID de la subcarpeta (ej. 'planes' o 'proyectos') const query = `'${rootFolderId}' in parents and name = '${subfolderName}' and mimeType = 'application/vnd.google-apps.folder' and trashed = false`; const res = await fetch(`https://www.googleapis.com/drive/v3/files?q=${encodeURIComponent(query)}`, { headers: { Authorization: `Bearer ${accessToken}` } }); const data = await res.json(); if (!data.files || data.files.length === 0) { throw new Error(`Subcarpeta ${subfolderName} no encontrada.`); } const subfolderId = data.files[0].id; // Subir el archivo como texto plano (.md) const metadata = { name: fileName, parents: [subfolderId] }; const form = new FormData(); form.append('metadata', new Blob([JSON.stringify(metadata)], { type: 'application/json' })); form.append('file', new Blob([markdownContent], { type: 'text/markdown' })); const uploadResponse = await fetch('https://www.googleapis.com/upload/drive/v3/files?uploadType=multipart', { method: 'POST', headers: { Authorization: `Bearer ${accessToken}` }, body: form }); if (uploadResponse.ok) { console.log(`Archivo ${fileName} guardado correctamente en ${subfolderName}.`); alert("Guardado exitosamente en Google Drive."); } else { console.error("Error al subir archivo:", await uploadResponse.text()); } } catch (error) { console.error("Error en la operación de guardado:", error); } }
